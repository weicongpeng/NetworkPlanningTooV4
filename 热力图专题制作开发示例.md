@app.post("/api/upload/tab")
async def upload_tab(
    file: UploadFile = File(...),
    weight_field: str = Form(default="value"),
    radius: int = Form(default=25),
    blur: int = Form(default=15),
    max_intensity: float = Form(default=0.0),
):
    """上传TAB文件并返回热力图GeoJSON数据"""
    if not file.filename.lower().endswith(('.tab', '.zip')):
        raise HTTPException(status_code=400, detail="请上传 .tab 或 .zip 文件")

    # 保存文件
    file_id = str(uuid.uuid4())[:8]
    file_path = f"data/{file_id}_{file.filename}"
    with open(file_path, "wb") as f:
        content = await file.read()
        f.write(content)

    # 解析TAB文件
    try:
        points = parse_tab_file(file_path, weight_field)
        geojson = generate_heatmap_geojson(points, radius, blur, max_intensity)
        return JSONResponse(content=geojson)
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"解析TAB文件失败: {str(e)}")

        f.write(await file.read())
    try:
        from osgeo import ogr
        ds = ogr.Open(file_path)
        layer = ds.GetLayer(0)
        layer_defn = layer.GetLayerDefn()
        fields = [layer_defn.GetFieldDefn(i).GetName() 
                  for i in range(layer_defn.GetFieldCount())]
        ds = None
        return {"fields": fields}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
    finally:
        if os.path.exists(file_path):
            os.remove(file_path)

3.3 TAB文件解析 services/tab_parser.py
from osgeo import ogr, osr
from typing import List, Dict, Optional
import zipfile
import os
import tempfile


def parse_tab_file(file_path: str, weight_field: str = "value") -> List[Dict]:
    """
    解析MapInfo TAB文件，提取点坐标和权重值
    
    参数:
        file_path: TAB文件路径（支持.zip压缩包）
        weight_field: 权重字段名
    
    返回:
        [{"lng": float, "lat": float, "weight": float}, ...]
    """
    # 如果是zip，先解压
    if file_path.lower().endswith('.zip'):
        file_path = extract_tab_zip(file_path)

    # 注册OGR驱动
    ogr.RegisterAll()

    ds = ogr.Open(file_path)
    if ds is None:
        raise ValueError(f"无法打开文件: {file_path}")

    layer = ds.GetLayer(0)
    
    # 检查坐标系，如果不是WGS84则转换
    source_srs = layer.GetSpatialRef()
    target_srs = osr.SpatialReference()
    target_srs.ImportFromEPSG(4326)  # WGS84
    
    transform = None
    if source_srs is not None:
        transform = osr.CoordinateTransformation(source_srs, target_srs)

    # 检查权重字段是否存在
    layer_defn = layer.GetLayerDefn()
    field_names = [layer_defn.GetFieldDefn(i).GetName() 
                   for i in range(layer_defn.GetFieldCount())]
    
    if weight_field not in field_names:
        print(f"警告: 字段 '{weight_field}' 不存在，可用字段: {field_names}")
        print("将使用默认权重值 1.0")
        weight_field = None

    points = []
    for feature in layer:
        geom = feature.GetGeometryRef()
        if geom is None:
            continue
            
        geom_type = geom.GetGeometryType()
        
        # 处理点要素
        if geom_type == ogr.wkbPoint:
            lng, lat = geom.GetX(), geom.GetY()
        # 处理面要素（取质心）
        elif geom_type in (ogr.wkbPolygon, ogr.wkbMultiPolygon):
            centroid = geom.Centroid()
            lng, lat = centroid.GetX(), centroid.GetY()
        else:
            continue

        # 坐标转换
        if transform:
            lng, lat, _ = transform.TransformPoint(lng, lat)

        # 获取权重值
        weight = 1.0
        if weight_field:
            try:
                weight = float(feature.GetField(weight_field))
            except (TypeError, ValueError):
                weight = 1.0

        points.append({
            "lng": lng,
            "lat": lat,
            "weight": weight
        })

    ds = None  # 关闭数据源
    return points


def extract_tab_zip(zip_path: str) -> str:
    """解压TAB的zip包，返回.tab文件路径"""
    extract_dir = os.path.splitext(zip_path)[0]
    os.makedirs(extract_dir, exist_ok=True)
    
    with zipfile.ZipFile(zip_path, 'r') as zf:
        zf.extractall(extract_dir)
    
    # 查找.tab文件
    for root, dirs, files in os.walk(extract_dir):
        for f in files:
            if f.lower().endswith('.tab'):
                return os.path.join(root, f)
    
    raise ValueError("ZIP包中未找到 .tab 文件")
3.4 表格数据解析 services/table_parser.py
import pandas as pd
from typing import List, Dict
import os


def parse_table_data(
    file_path: str,
    lat_field: str = "lat",
    lng_field: str = "lng",
    weight_field: str = "value"
) -> List[Dict]:
    """
    解析表格数据（CSV/Excel），提取坐标和权重
    
    参数:
        file_path: 文件路径
        lat_field: 纬度字段名
        lng_field: 经度字段名
        weight_field: 权重字段名
    
    返回:
        [{"lng": float, "lat": float, "weight": float}, ...]
    """
    ext = os.path.splitext(file_path)[1].lower()
    
    if ext == '.csv':
        df = pd.read_csv(file_path, encoding='utf-8-sig')
    elif ext in ('.xlsx', '.xls'):
        df = pd.read_excel(file_path)
    else:
        raise ValueError(f"不支持的文件格式: {ext}")

    # 验证必要字段
    missing_fields = []
    for field in [lat_field, lng_field]:
        if field not in df.columns:
            missing_fields.append(field)
    
    if missing_fields:
        raise ValueError(
            f"缺少必要字段: {missing_fields}。"
            f"可用字段: {list(df.columns)}"
        )

    # 如果权重字段不存在，默认权重为1
    if weight_field not in df.columns:
        df[weight_field] = 1.0

    # 清洗数据
    df = df.dropna(subset=[lat_field, lng_field])
    df[weight_field] = pd.to_numeric(df[weight_field], errors='coerce').fillna(0)

    # 构建点列表
    points = []
    for _, row in df.iterrows():
        points.append({
            "lng": float(row[lng_field]),
            "lat": float(row[lat_field]),
            "weight": float(row[weight_field])
        })

    return points
3.5 热力图工具 services/heatmap_utils.py
from typing import List, Dict
import json


def generate_heatmap_geojson(
    points: List[Dict],
    radius: int = 25,
    blur: int = 15,
    max_intensity: float = 0.0
) -> Dict:
    """
    将点数据转换为前端热力图所需的GeoJSON格式
    
    参数:
        points: 点数据列表 [{"lng", "lat", "weight"}]
        radius: 热力图点半径（像素）
        blur: 模糊度
        max_intensity: 最大强度（0表示自动计算）
    
    返回:
        GeoJSON FeatureCollection
    """
    if not points:
        return {
            "type": "FeatureCollection",
            "features": [],
            "metadata": {"count": 0, "radius": radius, "blur": blur}
        }

    # 自动计算最大强度
    if max_intensity <= 0:
        max_intensity = max(p["weight"] for p in points) if points else 1.0

    features = []
    for p in points:
        features.append({
            "type": "Feature",
            "geometry": {
                "type": "Point",
                "coordinates": [p["lng"], p["lat"]]
            },
            "properties": {
                "weight": p["weight"]
            }
        })

    return {
        "type": "FeatureCollection",
        "features": features,
        "metadata": {
            "count": len(features),
            "radius": radius,
            "blur": blur,
            "max_intensity": max_intensity
        }
    }


def generate_heatmap_grid(
    points: List[Dict],
    grid_size: float = 0.01,
    bounds: tuple = None
) -> Dict:
    """
    将点数据聚合为网格热力图（适合大数据量场景）
    
    参数:
        points: 点数据
        grid_size: 网格大小（度）
        bounds: 边界范围 (min_lng, min_lat, max_lng, max_lat)
    """
    from collections import defaultdict
    
    grid = defaultdict(float)
    
    for p in points:
        grid_x = int(p["lng"] / grid_size)
        grid_y = int(p["lat"] / grid_size)
        grid[(grid_x, grid_y)] += p["weight"]

    features = []
    for (gx, gy), weight in grid.items():
        center_lng = (gx + 0.5) * grid_size
        center_lat = (gy + 0.5) * grid_size
        features.append({
            "type": "Feature",
            "geometry": {
                "type": "Point",
                "coordinates": [center_lng, center_lat]
            },
            "properties": {"weight": weight}
        })

    return {
        "type": "FeatureCollection",
        "features": features,
        "metadata": {"count": len(features), "grid_size": grid_size}
    }

        // ========== 生成热力图 ==========
        async function generateHeatmap() {
            const formData = new FormData();
            let url = '';
            
            if (currentDataSource === 'tab') {
                const file = document.getElementById('tabFile').files[0];
                if (!file) { showStatus('请先选择TAB文件', 'error'); return; }
                formData.append('file', file);
                formData.append('weight_field', document.getElementById('tabWeightField').value);
                url = '/api/upload/tab';
            } else {
                const file = document.getElementById('tableFile').files[0];
                if (!file) { showStatus('请先选择表格文件', 'error'); return; }
                formData.append('file', file);
                formData.append('lat_field', document.getElementById('latField').value);
                formData.append('lng_field', document.getElementById('lngField').value);
                formData.append('weight_field', document.getElementById('tableWeightField').value);
                url = '/api/upload/table';
            }
            
            formData.append('radius', document.getElementById('radius').value);
            formData.append('blur', document.getElementById('blur').value);
            formData.append('max_intensity', document.getElementById('maxIntensity').value);

            showStatus('⏳ 正在处理数据...', 'success');

            try {
                const response = await fetch(url, {
                    method: 'POST',
                    body: formData
                });
                
                if (!response.ok) {
                    const err = await response.json();
                    throw new Error(err.detail || '请求失败');
                }
                
                const geojson = await response.json();
                renderHeatmap(geojson);
                showStatus(`✅ 成功渲染 ${geojson.metadata.count} 个数据点`, 'success');
                
            } catch (error) {
                showStatus(`❌ 错误: ${error.message}`, 'error');
            }
        }

        // ========== 渲染热力图 ==========
        function renderHeatmap(geojson) {
            // 清除已有热力图层
            if (heatLayer) {
                map.removeLayer(heatLayer);
            }

            const radius = parseInt(document.getElementById('radius').value);
            const blur = parseInt(document.getElementById('blur').value);
            const maxIntensity = parseFloat(document.getElementById('maxIntensity').value) || geojson.metadata.max_intensity;
            const scheme = document.getElementById('colorScheme').value;

            // 转换为 Leaflet.heat 格式: [lat, lng, intensity]
            const heatData = geojson.features.map(f => [
                f.geometry.coordinates[1],  // lat
                f.geometry.coordinates[0],  // lng
                f.properties.weight / maxIntensity  // 归一化强度
            ]);

            // 创建热力图层
            heatLayer = L.heatLayer(heatData, {
                radius: radius,
                blur: blur,
                maxZoom: 17,
                max: 1.0,
                gradient: colorSchemes[scheme]
            });

            heatLayer.addTo(map);

            // 自动缩放到数据范围
            if (heatData.length > 0) {
                const bounds = L.latLngBounds(heatData.map(d => [d[0], d[1]]));
                map.fitBounds(bounds, { padding: [50, 50] });
            }

            // 显示图例
            document.getElementById('legend').style.display = 'block';
        }

        // ========== 清除热力图 ==========
        function clearHeatmap() {
            if (heatLayer) {
                map.removeLayer(heatLayer);
                heatLayer = null;
            }
            document.getElementById('legend').style.display = 'none';
            showStatus('已清除热力图', 'success');
        }

            el.textContent = msg;
            el.className = `status-msg ${type}`;
        }
