```typescript
// Haversine 公式计算两点距离
function getDistance(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const R = 6371000; // 地球半径（米）
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
            Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
            Math.sin(dLon/2) * Math.sin(dLon/2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
  return R * c;
}

// 计算多点总距离
function calculateTotalDistance(points: MeasurePoint[]): number {
  let total = 0;
  for (let i = 1; i < points.length; i++) {
    total += getDistance(
      points[i-1].lat, points[i-1].lng,
      points[i].lat, points[i].lng
    );
  }
  return total;
}
```

## 坐标转换系统

### 背景

中国境内必须做 **WGS84 ↔ GCJ02 转换**:
- **WGS84**: GPS 标准坐标，后端返回的数据使用
- **GCJ02**: 国测局坐标（火星坐标），高德地图使用，在中国境内有偏移

### 核心算法

```typescript
// WGS84 → GCJ02 (正向转换)
export function wgs84ToGcj02(lat: number, lng: number): [number, number] {
  // 1. 判断是否在中国境外 (境外不做转换)
  if (outOfChina(lat, lng)) return [lat, lng];
  
  // 2. 计算偏移量 (基于克拉索夫斯基椭球体参数)
  let dlat = transformLat(lng - 105.0, lat - 35.0);
  let dlng = transformLng(lng - 105.0, lat - 35.0);
  
  // 3. 将偏移量转换为经纬度增量
  const radlat = lat / 180.0 * Math.PI;
  let magic = Math.sin(radlat);
  magic = 1 - 0.00669342162296594323 * magic * magic;  // 偏心率平方
  const sqrtmagic = Math.sqrt(magic);
  dlat = (dlat * 180.0) / ((6378245.0 * (1 - 0.00669342162296594323)) / (magic * sqrtmagic) * Math.PI);
  dlng = (dlng * 180.0) / (6378245.0 / sqrtmagic * Math.cos(radlat) * Math.PI);
  
  // 4. 返回偏移后的坐标
  return [lat + dlat, lng + dlng];
}

// GCJ02 → WGS84 (逆向转换,迭代逼近法)
export function gcj02ToWgs84(lat: number, lng: number): [number, number] {
  if (outOfChina(lat, lng)) return [lat, lng];
  
  let wgsLat = lat;
  let wgsLng = lng;
  
  // 迭代 5 次逼近真实值
  for (let i = 0; i < 5; i++) {
    const [gcjLat, gcjLng] = wgs84ToGcj02(wgsLat, wgsLng);
    const dLat = gcjLat - lat;
    const dLng = gcjLng - lng;
    wgsLat -= dLat;
    wgsLng -= dLng;
    if (Math.abs(dLat) < 1e-8 && Math.abs(dLng) < 1e-8) break;
  }
  
  return [wgsLat, wgsLng];
}
```

### 中国境外判断

```typescript
function outOfChina(lat: number, lng: number): boolean {
  return (lng < 72.004 || lng > 137.8347) || (lat < 0.8293 || lat > 55.8271);
}
```

**范围:** 经度 72.004°-137.8347°, 纬度 0.8293°-55.8271°

### 转换时机

1. **后端数据导入**: `dataParser.ts` 解析时立即转换
2. **本地文件导入**: `streamingParser.ts` / `asyncParser.ts` 解析时转换
3. **导航定位**: `navigationService.ts` 获取 GPS 坐标后转换

**关键原则:** 所有进入地图渲染的坐标必须是 GCJ02

      // - 解析经纬度
      // - WGS84 → GCJ02 转换

    // 5.3 定期让出主线程 (每 100 行)

### 异步解析 (asyncParser.ts)

**适用场景:** 中等文件 (1MB-5MB)

**核心机制:**
```typescript
// 1. 分片处理数组
export async function processInChunks<T, R>(
  items: T[],
  processor: (item: T, index: number) => R | null,
  options: AsyncParseOptions = {}
): Promise<R[]> {
  const { chunkSize = 100, yieldInterval = 50 } = options;
  const results: R[] = [];
  
  for (let i = 0; i < items.length; i++) {
    const result = processor(items[i], i);
    if (result !== null) results.push(result);
    
    // 定期让出主线程 (每 50 行)
    if (i % yieldInterval === 0 && i > 0) {
      await yieldToMainThread();  // setImmediate 或 setTimeout
    }
    
    // 进度回调 (每 100 行)
    if (i % chunkSize === 0) {
      onProgress?.({ processed: i, total: items.length, percentage });
    }
  }
  
  return results;
}

// 2. 等待交互空闲
const waitForInteraction = () => new Promise<void>(resolve => {
  InteractionManager.runAfterInteractions(() => resolve());
});

// 3. 让出主线程
const yieldToMainThread = () => new Promise<void>(resolve => {
  setImmediate(resolve);  // React Native 中优先使用 setImmediate
});
```

### 解析策略选择

```typescript
// cacheService.ts importFile() 中的决策树
if (isLargeFile(fileSize)) {
  // 大文件: 流式解析
  try {
    result = await parseFileStreaming(arrayBuffer, fileName, {
      batchSize: 1000,
      onProgress,
      onBatch: async (features) => {
        allFeatures.push(...features);
        await yieldToUI();
      }
    });
  } catch (error) {
    // 流式失败: 回退到异步解析
    result = await parseFileAsync(arrayBuffer, fileName, {
      chunkSize: 500,
      yieldInterval: 100,
      onProgress
    });
  }
} else if (fileSize > 1 * 1024 * 1024) {
  // 中等文件: 异步解析
  result = await parseFileAsync(arrayBuffer, fileName, options);
} else {
  // 小文件: 同步解析
  result = parseImportedFile(arrayBuffer, fileName);
}
```

### 防 ANR 机制

**JS 线程让出策略:**
```typescript
const yieldToUI = () => new Promise<void>(resolve => setTimeout(resolve, 0));
```

**使用时机:**
1. 每处理 50-100 行数据后让出一次
2. 每批处理完成后让出一次
3. 文件读取完成后让出一次
4. 坐标转换完成后让出一次

**效果:** 保持 UI 响应，避免 Application Not Responding (ANR) 弹窗
